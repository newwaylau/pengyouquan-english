// 武器攻击结算伪代码
function weaponAttack(hero, target) {
  const weapon = hero.getWeapon();
  if (!weapon || weapon.durability <= 0) return;
  
  // 英雄攻击力 = 基础值 + 武器攻击力 + Buff
  const totalAtk = hero.baseAttack + weapon.attack + sumBuff(hero, 'attack');
  
  // 造成伤害
  damageTarget(target, totalAtk);
  
  // 耐久度消耗
  weapon.durability -= 1;
  if (weapon.durability <= 0) {
    hero.destroyWeapon();
    emitEvent('weapon_broken', { weapon });
  }
  
  // 攻击动画
  playAnimation('hero_attack', { weapon, target });
}

// 计时器实现
const TURN_TIMER = {
  total: 75000,
  extra: 30000,
  ropeStart: 15000,
  autoEnd: 0,
  firstTurnBonus: 15000,
};

const [timeLeft, setTimeLeft] = useState(75000);
const [isRopeBurning, setIsRopeBurning] = useState(false);

// 每秒更新
useEffect(() => {
  if (timeLeft <= 0) { endTurn(); return; }
  const timer = setInterval(() => {
    setTimeLeft(prev => {
      if (prev <= TURN_TIMER.ropeStart && !isRopeBurning) {
        setIsRopeBurning(true);
        playSound('rope_burn');
      }
      return prev - 1000;
    });
  }, 1000);
  return () => clearInterval(timer);
}, [timeLeft]);

2.1 计算参数（10维向量）

每张卡牌的计算参数向量 P = [p₀, p₁, p₂, p₃, p₄, p₅, p₆, p₇, p₈, p₉] 决定了动画模板选择和参数调优：

索引参数名范围数据来源描述
p₀raceId0–9 cards.race 种族ID：0=无, 1=龙, 2=野兽, 3=人类, 4=异鬼, 5=巨人, 6=元素, 7=恶魔, 8=机械, 9=神话。决定动画的基础风格倾向——龙族偏火，异鬼偏冰等。
p₁elementId0–9 cards.element 元素ID：0=无, 1=火, 2=冰, 3=暗影, 4=圣光, 5=自然, 6=雷电, 7=金属, 8=毒, 9=水。直接决定粒子颜色主题。
p₂keywordMask0–127 cards.keywords 关键词位掩码：bit0=嘲讽, bit1=圣盾, bit2=亡语, bit3=战吼, bit4=潜行, bit5=突袭, bit6=冲锋。掩码影响动画的入场/出场特效强度。
p₃costTier0–3 cards.cost 费用层级：0=0-1费, 1=2-3费, 2=4-6费, 3=7-9费。费用越高，动画越华丽，粒子越多。
p₄rarityLv1–5 cards.rarity 稀有度等级：1=普通, 2=稀有, 3=史诗, 4=传说, 5=特殊。稀有度乘数影响所有参数。
p₅atkWeight0.0–1.0 cards.attack 攻击力权重：attack/maxAttack（最大攻击10）。高攻击=偏向攻击型动画（冲击/爆炸）。
p₆hpWeight0.0–1.0 cards.health 生命值权重：health/maxHealth（最大生命10）。高生命=偏向防御型动画（护盾/光环）。
p₇cardTypeId0–3 cards.card_type 卡牌类型：0=随从, 1=法术, 2=装备, 3=场地。法术卡偏瞬发光效，随从卡偏阶段性动画。
p₈factionId0–8 cards.faction 阵营ID：0=中立, 1=史塔克, 2=兰尼斯特, 3=坦格利安, 4=守夜人, 5=异鬼, 6=马泰尔, 7=葛雷乔伊, 8=提利尔。阵营影响颜色倾向。
p₉speedPref0.3–1.2 cost×rarity 速度偏好：0.3=极慢（史诗/传说大卡）, 1.2=极快（低费冲锋）。低费卡动画更快。计算方式：1.2 - costTier×0.2 - (rarityLv-1)×0.08。 
// 从数据库字段计算10维向量
function computeAnimationVector(card: Card): number[] {
  const raceMap: Record = { 'none':0,'dragon':1,'beast':2,'human':3,'undead':4,'giant':5,'elemental':6,'demon':7,'mech':8,'mythic':9 };
  const elemMap: Record = { 'none':0,'fire':1,'ice':2,'shadow':3,'holy':4,'nature':5,'thunder':6,'metal':7,'poison':8,'water':9 };
  const typeMap: Record = { 'minion':0,'spell':1,'equipment':2,'location':3 };
  const factionMap: Record = { 'neutral':0,'stark':1,'lannister':2,'targaryen':3,'nightwatch':4,'whitewalker':5,'martell':6,'greyjoy':7,'tyrell':8 };
  const rarityMap: Record = { 'common':1,'rare':2,'epic':3,'legendary':4,'special':5 };
  
  const keywordMask = (card.keywords?.includes('taunt')?1:0) | (card.keywords?.includes('divine_shield')?2:0)
    | (card.keywords?.includes('deathrattle')?4:0) | (card.keywords?.includes('battlecry')?8:0)
    | (card.keywords?.includes('stealth')?16:0) | (card.keywords?.includes('rush')?32:0)
    | (card.keywords?.includes('charge')?64:0);
  
  const costTier = card.cost <= 1 ? 0 : card.cost <= 3 ? 1 : card.cost <= 6 ? 2 : 3;
  const atkW = Math.min((card.attack || 0) / 10, 1.0);
  const hpW = Math.min((card.health || 0) / 10, 1.0);
  const speed = 1.2 - costTier * 0.2 - (rarityMap[card.rarity] - 1) * 0.08;
  
  return [
    raceMap[card.race] ?? 0,
    elemMap[card.element] ?? 0,
    keywordMask,
    costTier,
    rarityMap[card.rarity] ?? 1,
    atkW,
    hpW,
    typeMap[card.card_type] ?? 0,
    factionMap[card.faction] ?? 0,
    Math.max(0.3, Math.min(1.2, speed))
  ];
}

2.3 参数→模板映射算法

10维参数向量通过加权距离计算选择最匹配的动画模板，然后进一步调优模板参数：

// ============================================================
// 参数→模板选择算法
// ============================================================

// 每个模板有一个"特征向量" T 和权重向量 W
interface TemplateProfile {
  id: number;
  name: string;
  vector: number[];    // 10维特征向量（与P向量同空间）
  weights: number[];   // 各维度权重（某些维度对该模板更重要）
  params: {
    duration: number;        // 基础持续时间(ms)
    particleCount: number;   // 基础粒子数
    shakeIntensity: number;  // 震动强度(0-10)
    colorTheme: string;      // CSS颜色主题
    soundEffect: string;     // 音效ID
    canvasEffect: string;    // Canvas特效类型
  };
}

// 选择最匹配的模板
function selectTemplate(P: number[]): TemplateProfile {
  let bestId = -1;
  let bestScore = -Infinity;
  
  for (const template of allTemplates) {
    // 加权余弦相似度
    let dotProduct = 0;
    let normP = 0;
    let normT = 0;
    
    for (let i = 0; i < 10; i++) {
      const w = template.weights[i];
      const pVal = P[i] * w;
      const tVal = template.vector[i] * w;
      dotProduct += pVal * tVal;
      normP += pVal * pVal;
      normT += tVal * tVal;
    }
    
    const similarity = dotProduct / (Math.sqrt(normP) * Math.sqrt(normT) + 0.0001);
    
    // elementId (p₁) 权重加倍——元素是动画风格的主要决定因素
    const elementBonus = (P[1] === template.vector[1]) ? 0.3 : 0;
    // keywordMask (p₂) 部分匹配加分
    const keywordMatch = bitwiseOverlap(P[2], template.vector[2]) * 0.1;
    
    const score = similarity * 0.6 + elementBonus + keywordMatch;
    
    if (score > bestScore) {
      bestScore = score;
      bestId = template.id;
    }
  }
  
  return allTemplates[bestId];
}

// 选择模板后，调优参数
function tuneParameters(P: number[], template: TemplateProfile): TunedParams {
  const rarityMult = [1.0, 1.3, 1.8, 2.5, 4.0][P[4] - 1];  // 稀有度乘数
  const costMult = 0.8 + P[3] * 0.3;                        // 费用乘数
  const speedFactor = P[9];                                  // 速度偏好
  
  return {
    duration: Math.round(template.params.duration * (1 / speedFactor) * costMult),
    particleCount: Math.round(template.params.particleCount * rarityMult * (0.7 + P[5] * 0.3)),
    shakeIntensity: Math.min(10, Math.round(template.params.shakeIntensity * rarityMult * 0.7)),
    colorTheme: blendColors(template.params.colorTheme, getFactionColor(P[8])),
    soundEffect: template.params.soundEffect,
    canvasEffect: template.params.canvasEffect,
    rarityGlow: P[4] >= 3,         // 史诗及以上添加辉光
    screenFlash: P[4] >= 4,         // 传说及以上添加全屏闪
    extraParticles: P[5] > 0.7,     // 高攻击额外粒子
    slowMotion: P[4] >= 4 && speedFactor < 0.6,  // 传说+慢速→慢动作
  };
}

// 计算位掩码重叠
function bitwiseOverlap(a: number, b: number): number {
  const overlap = a & b;
  let count = 0;
  for (let i = 0; i < 7; i++) {
    if (overlap & (1 << i)) count++;
  }
  return count / 7;
}

伤害计算公式

// ==========================================
// 最终伤害 = (基础伤害 + 力量) × 易伤 × 虚弱 × 缠绕 × 标记
// ==========================================
function calculateDamage(
  baseDamage: number,      // 卡牌基础伤害值
  strength: number,        // 攻击方力量（默认0）
  isVulnerable: boolean,   // 目标是否易伤
  isWeak: boolean,         // 攻击方是否虚弱
  slowStacks: number,      // 目标缠绕层数
  isMarked: boolean,       // 目标是否被标记
  cardsPlayedThisTurn: number  // 本回合已出牌数
): number {
  let dmg = baseDamage + strength;        // 力量加成
  if (isVulnerable) dmg *= 1.5;           // 易伤+50%
  if (isWeak) dmg *= 0.75;                // 虚弱-25%
  dmg *= (1 + slowStacks * 0.1 * cardsPlayedThisTurn); // 缠绕递增
  if (isMarked) dmg *= 1.5;               // 标记+50%
  return Math.max(1, Math.round(dmg));     // 最小1点伤害
}

// ==========================================
// 格挡：分段消耗，溢出穿透，回合结束移除
// ==========================================
interface BlockSystem {
  currentBlock: number;      // 当前格挡值
  maxBlockPerTurn: number;   // 单回合最大格挡（默认999，可被限制）
  
  // 获得格挡（可叠加）
  addBlock(amount: number): void;
  
  // 消耗格挡抵挡伤害
  // 返回：实际受到的生命伤害（穿透伤害）
  consumeBlock(damage: number): number {
    const blocked = Math.min(this.currentBlock, damage);
    const penetrated = damage - blocked;
    this.currentBlock -= blocked;
    return penetrated;  // 穿透的生命伤害
  }
  
  // 回合结束清除
  endTurn(): void {
    this.currentBlock = 0;
  }
}

抽牌/弃牌/洗牌机制

// ==========================================
// 完整三区循环
// ==========================================
class CardDeck {
  drawPile: Card[] = [];       // 抽牌堆
  hand: Card[] = [];            // 手牌（上限10张）
  discardPile: Card[] = [];     // 弃牌堆
  exhaustPile: Card[] = [];     // 消耗区（移出战斗）

  // 摸牌：从抽牌堆顶取N张→手牌
  drawCards(count: number): Card[] {
    const drawn: Card[] = [];
    for (let i = 0; i < count; i++) {
      if (this.drawPile.length === 0) {
        this.shuffle();  // 弃牌堆→洗牌→抽牌堆
      }
      if (this.drawPile.length === 0) break; // 仍为空→无牌可抽
      const card = this.drawPile.pop()!;
      if (this.hand.length < 10) {
        this.hand.push(card);
        drawn.push(card);
      } else {
        // 手牌满→直接进入弃牌堆
        this.discardPile.push(card);
      }
    }
    return drawn;
  }

  // 洗牌：弃牌堆→抽牌堆（Fisher-Yates洗牌）
  shuffle(): void {
    this.drawPile.push(...this.discardPile);
    this.discardPile = [];
    // Fisher-Yates 洗牌
    for (let i = this.drawPile.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [this.drawPile[i], this.drawPile[j]] = [this.drawPile[j], this.drawPile[i]];
    }
  }

  // 回合结束：所有手牌→弃牌堆
  endTurn(): void {
    this.discardPile.push(...this.hand);
    this.hand = [];
  }

  // 回合开始：摸5张牌
  startTurn(): void {
    this.drawCards(5);
  }
}
